cmake_minimum_required(VERSION 3.24)
project(DeribitGrid VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
option(GRID_BUILD_GUI "Build the Qt desktop application" ON)
add_library(grid_core src/grid.cpp)
target_include_directories(grid_core PUBLIC src)
if(MSVC)
  add_compile_options(/utf-8 /W4 /permissive- /EHsc)
  target_compile_options(grid_core PRIVATE /utf-8 /W4 /permissive- /EHsc)
else()
  add_compile_options(-Wall -Wextra -Wpedantic)
endif()
include(CTest)
if(BUILD_TESTING)
  add_executable(grid_tests tests/grid_tests.cpp)
  target_link_libraries(grid_tests PRIVATE grid_core)
  add_test(NAME grid_scenarios COMMAND grid_tests)
endif()
if(GRID_BUILD_GUI)
  set(CMAKE_INSTALL_BINDIR ".")
  find_package(Qt6 6.8 REQUIRED COMPONENTS Widgets WebSockets Network Sql Test)
  qt_standard_project_setup()
  qt_add_executable(DeribitGrid WIN32 src/main.cpp src/window.cpp src/client.cpp src/store.cpp)
  target_link_libraries(DeribitGrid PRIVATE grid_core Qt6::Widgets Qt6::WebSockets Qt6::Network Qt6::Sql)
  if(BUILD_TESTING)
    qt_add_executable(storage_tests tests/storage_tests.cpp src/store.cpp)
    target_link_libraries(storage_tests PRIVATE grid_core Qt6::Sql Qt6::Test)
    add_test(NAME storage_recovery COMMAND storage_tests)
    qt_add_executable(client_tests tests/client_tests.cpp src/client.cpp)
    target_include_directories(client_tests PRIVATE src)
    target_link_libraries(client_tests PRIVATE Qt6::WebSockets Qt6::Network Qt6::Test)
    add_test(NAME websocket_protocol COMMAND client_tests)
    qt_add_executable(window_tests tests/window_tests.cpp src/window.cpp src/client.cpp src/store.cpp)
    target_link_libraries(window_tests PRIVATE grid_core Qt6::Widgets Qt6::WebSockets Qt6::Network Qt6::Sql Qt6::Test)
    set_target_properties(window_tests PROPERTIES WIN32_EXECUTABLE FALSE)
    add_test(NAME desktop_execution COMMAND window_tests -platform offscreen)
    set_tests_properties(desktop_execution PROPERTIES TIMEOUT 50)
    qt_add_executable(testnet_probe tests/testnet_probe.cpp)
    target_link_libraries(testnet_probe PRIVATE Qt6::WebSockets Qt6::Network)
    set_target_properties(storage_tests client_tests testnet_probe PROPERTIES WIN32_EXECUTABLE FALSE)
  endif()
  install(TARGETS DeribitGrid RUNTIME DESTINATION .)
  qt_generate_deploy_app_script(TARGET DeribitGrid OUTPUT_SCRIPT deploy_script NO_UNSUPPORTED_PLATFORM_ERROR)
  install(SCRIPT ${deploy_script})
endif()
